# Unity Integration

## Introduction to Unity for Robotics

Unity is a powerful 3D development platform that has gained significant traction in robotics for creating simulation environments, visualization tools, and human-robot interaction interfaces. With its robust physics engine, high-quality rendering capabilities, and extensive asset library, Unity provides an excellent platform for developing sophisticated robotic applications, particularly for humanoid robots that require complex visual feedback and interaction design.

## Unity Robotics Setup

To get started with robotics in Unity, you'll need to set up the Unity Robotics Hub which includes packages for ROS communication, visualization, and simulation.

```csharp
// Example: Basic Unity robotics script structure
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Std;

public class UnityRobotController : MonoBehaviour
{
    // ROS connector
    private ROSConnection ros;

    // Robot components
    public GameObject robotBase;
    public Transform[] joints;
    public string robotName = "my_robot";

    void Start()
    {
        // Connect to ROS
        ros = ROSConnection.GetOrCreateInstance();
        ros.RegisterPublisher<JointStateMsg>("/joint_states");

        // Subscribe to commands
        ros.Subscribe<JointStateMsg>("/unity_joint_commands", OnJointCommandReceived);
    }

    void Update()
    {
        // Publish joint states
        PublishJointStates();
    }

    void OnJointCommandReceived(JointStateMsg cmd)
    {
        // Process joint commands from ROS
        for (int i = 0; i < joints.Length && i < cmd.position.Length; i++)
        {
            float targetAngle = (float)(cmd.position[i] * Mathf.Rad2Deg);
            joints[i].localRotation = Quaternion.Euler(0, targetAngle, 0);
        }
    }

    void PublishJointStates()
    {
        var jointState = new JointStateMsg();
        jointState.name = new string[joints.Length];
        jointState.position = new double[joints.Length];
        jointState.velocity = new double[joints.Length];
        jointState.effort = new double[joints.Length];

        for (int i = 0; i < joints.Length; i++)
        {
            jointState.name[i] = $"joint_{i}";
            jointState.position[i] = joints[i].localRotation.eulerAngles.y * Mathf.Deg2Rad;
            jointState.velocity[i] = 0; // Simplified
            jointState.effort[i] = 0;   // Simplified
        }

        ros.Publish("/joint_states", jointState);
    }
}
```

```csharp
using UnityEngine;
using Unity.Robotics.ROSTCPConnector;
using RosMessageTypes.Std;
using RosMessageTypes.Sensor;
using System.Collections;

public class UnityRobotController : MonoBehaviour
{
    // ROS connector
    private ROSConnection ros;

    // Robot components
    public GameObject robotBase;
    public Transform[] joints;
    public string robotName = "my_robot";

    // Camera for sensor simulation
    public Camera robotCamera;

    void Start()
    {
        // Connect to ROS
        ros = ROSConnection.GetOrCreateInstance();
        ros.RegisterPublisher<JointStateMsg>("/joint_states");
        ros.RegisterPublisher<ImageMsg>("/unity_camera/image_raw");

        // Subscribe to commands
        ros.Subscribe<JointStateMsg>("/unity_joint_commands", OnJointCommandReceived);

        // Start sensor publishing coroutine
        StartCoroutine(PublishCameraData());
    }

    void Update()
    {
        // Publish joint states at regular intervals
        if (Time.frameCount % 60 == 0) // Every 60 frames (about 1 Hz if running at 60 FPS)
        {
            PublishJointStates();
        }
    }

    void OnJointCommandReceived(JointStateMsg cmd)
    {
        // Process joint commands from ROS
        for (int i = 0; i < joints.Length && i < cmd.position.Length; i++)
        {
            float targetAngle = (float)(cmd.position[i] * Mathf.Rad2Deg);
            joints[i].localRotation = Quaternion.Euler(0, targetAngle, 0);
        }
    }

    void PublishJointStates()
    {
        var jointState = new JointStateMsg();
        jointState.header = new std_msgs.HeaderMsg();
        jointState.header.stamp = new builtin_interfaces.TimeMsg();
        jointState.header.stamp.sec = (int)Time.time;
        jointState.header.stamp.nanosec = (uint)((Time.time % 1) * 1e9);
        jointState.header.frame_id = robotName;

        jointState.name = new string[joints.Length];
        jointState.position = new double[joints.Length];
        jointState.velocity = new double[joints.Length];
        jointState.effort = new double[joints.Length];

        for (int i = 0; i < joints.Length; i++)
        {
            jointState.name[i] = $"joint_{i}";
            jointState.position[i] = joints[i].localRotation.eulerAngles.y * Mathf.Deg2Rad;
            jointState.velocity[i] = 0; // Simplified
            jointState.effort[i] = 0;   // Simplified
        }

        ros.Publish("/joint_states", jointState);
    }

    IEnumerator PublishCameraData()
    {
        while (true)
        {
            yield return new WaitForSeconds(0.1f); // 10 Hz

            if (robotCamera != null)
            {
                // Capture camera image and convert to ROS message
                var imageMsg = CaptureCameraImage(robotCamera);
                if (imageMsg != null)
                {
                    ros.Publish("/unity_camera/image_raw", imageMsg);
                }
            }
        }
    }

    ImageMsg CaptureCameraImage(Camera cam)
    {
        // Render texture approach for camera capture
        RenderTexture currentRT = RenderTexture.active;
        RenderTexture.active = cam.targetTexture;

        cam.Render();

        Texture2D imageTex = new Texture2D(cam.targetTexture.width, cam.targetTexture.height);
        imageTex.ReadPixels(new Rect(0, 0, cam.targetTexture.width, cam.targetTexture.height), 0, 0);
        imageTex.Apply();

        RenderTexture.active = currentRT;

        // Convert Texture2D to ROS Image message
        // This is a simplified example - actual implementation would require proper encoding
        var imageMsg = new ImageMsg();
        imageMsg.height = (uint)imageTex.height;
        imageMsg.width = (uint)imageTex.width;
        imageMsg.encoding = "rgb8";
        imageMsg.is_bigendian = 0;
        imageMsg.step = (uint)(imageTex.width * 3); // 3 bytes per pixel for RGB

        // In a real implementation, you would encode the image data properly
        // For this example, we'll just create a placeholder
        imageMsg.data = new byte[imageMsg.step * imageMsg.height];

        return imageMsg;
    }
}
```

## Humanoid Robot Models in Unity

Unity provides excellent tools for creating and animating humanoid robot models with complex joint structures:

```csharp
// Example: Humanoid robot controller with inverse kinematics
using UnityEngine;

public class HumanoidRobotController : MonoBehaviour
{
    [Header("Body Parts")]
    public Transform pelvis;
    public Transform spine;
    public Transform chest;
    public Transform neck;
    public Transform head;

    [Header("Arms")]
    public Transform leftShoulder, leftUpperArm, leftForearm, leftHand;
    public Transform rightShoulder, rightUpperArm, rightForearm, rightHand;

    [Header("Legs")]
    public Transform leftThigh, leftCalf, leftFoot;
    public Transform rightThigh, rightCalf, rightFoot;

    [Header("IK Targets")]
    public Transform leftHandTarget, rightHandTarget;
    public Transform leftFootTarget, rightFootTarget;

    [Header("Parameters")]
    public float moveSpeed = 2.0f;
    public float turnSpeed = 100.0f;

    private Animator animator;
    private bool useIK = true;

    void Start()
    {
        animator = GetComponent<Animator>();
    }

    void Update()
    {
        HandleMovement();
    }

    void HandleMovement()
    {
        // Get input
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");

        // Move and rotate
        Vector3 movement = new Vector3(horizontal, 0, vertical) * moveSpeed * Time.deltaTime;
        transform.Translate(movement, Space.Self);

        float turn = Input.GetAxis("Mouse X") * turnSpeed * Time.deltaTime;
        transform.Rotate(0, turn, 0);
    }

    void OnAnimatorIK(int layerIndex)
    {
        if (!useIK) return;

        // Set up IK for hands
        animator.SetIKPositionWeight(AvatarIKGoal.LeftHand, 1);
        animator.SetIKPosition(AvatarIKGoal.LeftHand, leftHandTarget.position);
        animator.SetIKRotationWeight(AvatarIKGoal.LeftHand, 1);
        animator.SetIKRotation(AvatarIKGoal.LeftHand, leftHandTarget.rotation);

        animator.SetIKPositionWeight(AvatarIKGoal.RightHand, 1);
        animator.SetIKPosition(AvatarIKGoal.RightHand, rightHandTarget.position);
        animator.SetIKRotationWeight(AvatarIKGoal.RightHand, 1);
        animator.SetIKRotation(AvatarIKGoal.RightHand, rightHandTarget.rotation);

        // Set up IK for feet
        animator.SetIKPositionWeight(AvatarIKGoal.LeftFoot, 1);
        animator.SetIKPosition(AvatarIKGoal.LeftFoot, leftFootTarget.position);
        animator.SetIKRotationWeight(AvatarIKGoal.LeftFoot, 1);
        animator.SetIKRotation(AvatarIKGoal.LeftFoot, leftFootTarget.rotation);

        animator.SetIKPositionWeight(AvatarIKGoal.RightFoot, 1);
        animator.SetIKPosition(AvatarIKGoal.RightFoot, rightFootTarget.position);
        animator.SetIKRotationWeight(AvatarIKGoal.RightFoot, 1);
        animator.SetIKRotation(AvatarIKGoal.RightFoot, rightFootTarget.rotation);
    }
}
```

## Physics Simulation in Unity

Unity's physics engine provides realistic simulation capabilities for robotic applications:

```csharp
// Example: Physics-based robot simulation
using UnityEngine;

public class PhysicsRobotController : MonoBehaviour
{
    [Header("Physics Parameters")]
    public float motorForce = 100f;
    public float maxJointVelocity = 5f;
    public float gravityScale = 1f;

    [Header("Joints")]
    public ConfigurableJoint[] joints;
    public Rigidbody[] linkBodies;

    [Header("Sensors")]
    public Transform centerOfMass;
    public float[] jointPositions;
    public float[] jointVelocities;

    void Start()
    {
        // Set up center of mass
        if (GetComponent<Rigidbody>() != null && centerOfMass != null)
        {
            GetComponent<Rigidbody>().centerOfMass = centerOfMass.localPosition;
        }

        // Initialize joint arrays
        if (joints.Length > 0)
        {
            jointPositions = new float[joints.Length];
            jointVelocities = new float[joints.Length];
        }
    }

    void FixedUpdate()
    {
        UpdatePhysics();
        UpdateSensors();
    }

    void UpdatePhysics()
    {
        // Apply forces to joints based on control input
        for (int i = 0; i < joints.Length; i++)
        {
            if (joints[i] != null)
            {
                // Example: Apply a simple control signal to each joint
                float controlSignal = GetJointControlSignal(i);

                // Apply force along the joint's primary axis
                Vector3 force = joints[i].axis * controlSignal * motorForce;
                joints[i].GetComponent<Rigidbody>().AddForceAtPosition(force, joints[i].transform.position);
            }
        }
    }

    void UpdateSensors()
    {
        // Update joint position and velocity sensors
        for (int i = 0; i < joints.Length; i++)
        {
            if (joints[i] != null)
            {
                // Get joint position (simplified)
                jointPositions[i] = joints[i].transform.localEulerAngles.y;

                // Get joint velocity (simplified)
                if (joints[i].connectedBody != null)
                {
                    jointVelocities[i] = joints[i].connectedBody.angularVelocity.y;
                }
            }
        }
    }

    float GetJointControlSignal(int jointIndex)
    {
        // This would typically come from a higher-level controller
        // For this example, we'll use a simple oscillating signal
        return Mathf.Sin(Time.time + jointIndex) * 0.5f;
    }

    // Get center of mass position
    public Vector3 GetCenterOfMassPosition()
    {
        if (centerOfMass != null)
        {
            return centerOfMass.position;
        }
        return transform.position;
    }

    // Get center of mass velocity
    public Vector3 GetCenterOfMassVelocity()
    {
        if (GetComponent<Rigidbody>() != null)
        {
            return GetComponent<Rigidbody>().velocity;
        }
        return Vector3.zero;
    }
}
```

## VR/AR Integration for Human-Robot Interaction

Unity's support for VR and AR makes it ideal for immersive human-robot interaction:

```csharp
// Example: VR-based robot teleoperation
using UnityEngine;
using UnityEngine.XR;

public class VRTeleoperationController : MonoBehaviour
{
    [Header("VR Controllers")]
    public GameObject leftController;
    public GameObject rightController;

    [Header("Robot Mapping")]
    public Transform robotLeftHand;
    public Transform robotRightHand;

    [Header("Teleoperation Settings")]
    public bool enableTeleoperation = true;
    public float positionScale = 1.0f;

    void Update()
    {
        if (enableTeleoperation)
        {
            UpdateRobotFromVRControllers();
        }
    }

    void UpdateRobotFromVRControllers()
    {
        if (leftController != null && robotLeftHand != null)
        {
            // Map VR controller position to robot hand
            robotLeftHand.position = leftController.transform.position * positionScale;
            robotLeftHand.rotation = leftController.transform.rotation;
        }

        if (rightController != null && robotRightHand != null)
        {
            // Map VR controller position to robot hand
            robotRightHand.position = rightController.transform.position * positionScale;
            robotRightHand.rotation = rightController.transform.rotation;
        }
    }

    // Method to get controller input for robot control
    public Vector3 GetControllerPosition(int controllerIndex)
    {
        if (controllerIndex == 0 && leftController != null)
        {
            return leftController.transform.position;
        }
        else if (controllerIndex == 1 && rightController != null)
        {
            return rightController.transform.position;
        }
        return Vector3.zero;
    }

    public Quaternion GetControllerRotation(int controllerIndex)
    {
        if (controllerIndex == 0 && leftController != null)
        {
            return leftController.transform.rotation;
        }
        else if (controllerIndex == 1 && rightController != null)
        {
            return rightController.transform.rotation;
        }
        return Quaternion.identity;
    }
}
```

- Robust physics simulation capabilities
- Extensive asset library and development tools
- VR/AR support for immersive interaction
- ROS integration through the Unity ROS TCP Connector
